(function() {
    document.addEventListener('DOMContentLoaded', function() {
        try {
            console.log('[AUTO-LOGIN] Checking for sessionStorage credentials...');
            var raw = sessionStorage.getItem('polysaas_new_user_creds');
            console.log('[AUTO-LOGIN] sessionStorage raw:', raw);
            if (!raw) {
                console.log('[AUTO-LOGIN] No credentials found in sessionStorage');
                return;
            }
            var creds = JSON.parse(raw);
            console.log('[AUTO-LOGIN] Parsed creds:', creds);
            sessionStorage.removeItem('polysaas_new_user_creds');
            if (creds.u) {
                var uField = document.getElementById('id_login');
                if (uField) {
                    uField.value = creds.u;
                    console.log('[AUTO-LOGIN] Set username field');
                }
            }
            if (creds.p) {
                var pField = document.getElementById('id_password');
                if (pField) {
                    pField.value = creds.p;
                    console.log('[AUTO-LOGIN] Set password field');
                }
            }
            var banner = document.getElementById('ps-new-user-banner');
            if (banner) {
                banner.textContent = '✅ Your new account "' + (creds.u || '') + '" has been pre-filled. Please review and click Sign in.';
                banner.style.display = 'block';
            }
        } catch(e) {
            console.error('[AUTO-LOGIN] Error:', e);
        }
    });

    // After successful login, restore Mattermost credentials from sessionStorage (without team name)
    if (sessionStorage.getItem('polysaas_mm_creds')) {
        var mmCreds = JSON.parse(sessionStorage.getItem('polysaas_mm_creds'));
        sessionStorage.removeItem('polysaas_mm_creds');
        console.log('[AUTO-LOGIN] Restoring Mattermost credentials to session:', mmCreds);
        // Call API to restore Mattermost credentials to Django session
        fetch('/dose/api/restore-mm-credentials/', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'X-CSRFToken': document.querySelector('[name=csrfmiddlewaretoken]')?.value || ''
            },
            body: JSON.stringify(mmCreds),
            credentials: 'same-origin'
        }).then(function(resp) {
            console.log('[AUTO-LOGIN] Restore credentials response:', resp.status);
        }).catch(function(e) {
            console.error('[AUTO-LOGIN] Failed to restore Mattermost credentials:', e);
        });
    }
})();
</script>
